/**
 * doGet - Handle GET requests from the web app
 * This supports JSONP for cross-origin requests
 */
function doGet(e) {
  // Set CORS headers for all responses
  const output = ContentService.createTextOutput();
  
  // Parse parameters
  const params = e.parameter;
  const sheetId = params.ID;  
  const sheetName = params.SH;
  const functionName = params.FN;
  const callback = params.callback || '';
  
  // Log request information for debugging
  console.log('Request received: ' + JSON.stringify(params));
  
  let result;
  
  try {
    // Call the appropriate function based on the functionName parameter
    switch(functionName) {
      case 'readSheet':
        result = readSheet(sheetId, sheetName);
        break;
      case 'appendRow':
        const data = params.DATA;
        result = appendRow(sheetId, sheetName, data);
        break;
      case 'appendRowJSON':
        const jsonData = JSON.parse(decodeURIComponent(params.DATA));
        result = appendRowJSON(sheetId, sheetName, jsonData);
        break;
      default:
        result = { error: 'Invalid function name', function: functionName };
    }
  } catch (error) {
    console.error('Error in doGet: ' + error.toString());
    result = { 
      error: error.toString(),
      message: "An error occurred while processing the request"
    };
  }
  
  // Convert result to JSON string
  const jsonString = JSON.stringify(result);
  
  // Return the result as JSONP if callback is provided, otherwise as JSON
  if (callback) {
    // JSONP response with callback wrapping
    output.setContent(callback + '(' + jsonString + ')');
    output.setMimeType(ContentService.MimeType.JAVASCRIPT);
  } else {
    // Regular JSON response
    output.setContent(jsonString);
    output.setMimeType(ContentService.MimeType.JSON);
  }
  
  return output;
}

/**
 * Read entire sheet and return as array
 */
function readSheet(sheetId, sheetName) {
  try {
    const ss = SpreadsheetApp.openById(sheetId);
    const sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      throw new Error('Sheet not found: ' + sheetName);
    }
    
    const dataRange = sheet.getDataRange();
    const values = dataRange.getValues();
    
    return values;
  } catch (e) {
    throw new Error('Error reading sheet: ' + e.toString());
  }
}

/**
 * Append a row to the sheet using comma-separated values
 */
function appendRow(sheetId, sheetName, data) {
  try {
    const ss = SpreadsheetApp.openById(sheetId);
    const sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      throw new Error('Sheet not found: ' + sheetName);
    }
    
    // Split the data into array values
    // This handles CSV format with quoted strings
    const rowData = parseCSV(data);
    
    // Append the row
    sheet.appendRow(rowData);
    
    return {
      success: true,
      message: 'Row appended successfully'
    };
  } catch (e) {
    throw new Error('Error appending row: ' + e.toString());
  }
}

/**
 * Append a row to the sheet using JSON data
 * This is more reliable than CSV for complex data
 */
function appendRowJSON(sheetId, sheetName, jsonData) {
  try {
    const ss = SpreadsheetApp.openById(sheetId);
    const sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      throw new Error('Sheet not found: ' + sheetName);
    }
    
    // Get headers from the first row
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    
    // Create row data array based on headers
    const rowData = headers.map(header => {
      const key = header.trim().toLowerCase().replace(/\s+/g, '_');
      return jsonData[key] !== undefined ? jsonData[key] : '';
    });
    
    // Append the row
    sheet.appendRow(rowData);
    
    return {
      success: true,
      message: 'Row appended successfully using JSON data'
    };
  } catch (e) {
    Logger.log('Error in appendRowJSON: ' + e.toString());
    throw new Error('Error appending row: ' + e.toString());
  }
}

/**
 * Parse CSV string into array, handling quoted values
 */
function parseCSV(csvString) {
  let result = [];
  let current = '';
  let inQuotes = false;
  
  for (let i = 0; i < csvString.length; i++) {
    const char = csvString[i];
    
    if (char === '"') {
      if (inQuotes && csvString[i+1] === '"') {
        // Double quotes inside quotes - add a single quote
        current += '"';
        i++;
      } else {
        // Toggle quote mode
        inQuotes = !inQuotes;
      }
    } else if (char === ',' && !inQuotes) {
      // End of field
      result.push(current);
      current = '';
    } else {
      current += char;
    }
  }
  
  // Add the last field
  result.push(current);
  
  return result;
}
